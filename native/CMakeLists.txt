cmake_minimum_required(VERSION 3.25)
project(leafseamer_installation_peer LANGUAGES CXX)
if(NOT WIN32 OR NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h" OR NOT EXISTS "${NODE_LIBRARY}")
  message(FATAL_ERROR "windows_node_inputs_required")
endif()
add_library(leafseamer_installation_peer SHARED installation-peer.cc)
target_include_directories(leafseamer_installation_peer PRIVATE "${NODE_INCLUDE_DIR}")
target_compile_definitions(leafseamer_installation_peer PRIVATE NAPI_VERSION=8 BUILDING_NODE_EXTENSION _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_features(leafseamer_installation_peer PRIVATE cxx_std_17)
target_compile_options(leafseamer_installation_peer PRIVATE /W4 /WX /utf-8 /guard:cf /Brepro)
target_link_options(leafseamer_installation_peer PRIVATE /guard:cf /Brepro)
target_link_libraries(leafseamer_installation_peer PRIVATE "${NODE_LIBRARY}" advapi32 bcrypt normaliz mincore)
set_target_properties(leafseamer_installation_peer PROPERTIES PREFIX "" SUFFIX ".node" MSVC_RUNTIME_LIBRARY MultiThreaded)
file(WRITE "${CMAKE_BINARY_DIR}/compiler.json" "{\"id\":\"${CMAKE_CXX_COMPILER_ID}\",\"version\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"windows_sdk\":\"${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}\"}\n")
